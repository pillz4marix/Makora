(function() {
    function inicializarLightbox() {
        const lightbox = document.getElementById("custom-lightbox");
        const lightboxImg = document.getElementById("lightbox-img");
        const closeBtn = document.querySelector(".lightbox-close");
        const prevBtn = document.querySelector(".lightbox-prev");
        const nextBtn = document.querySelector(".lightbox-next");
        const counter = document.getElementById("lightbox-counter");
        
        const imagenes = Array.from(document.querySelectorAll("img")).filter(img => img.id !== "lightbox-img" && img.offsetParent !== null);
        let indiceActual = 0;

        if (!lightbox || !lightboxImg) return;

        imagenes.forEach(function (imagen, indice) {
            imagen.style.cursor = "zoom-in";
            imagen.addEventListener("click", function () {
                indiceActual = indice;
                actualizarImagen();
                lightbox.style.display = "block";
                document.body.style.overflow = "hidden";
            });
        });

        function actualizarImagen() {
            if (imagenes[indiceActual]) {
                lightboxImg.src = imagenes[indiceActual].src;
                if (counter) {
                    counter.textContent = (indiceActual + 1) + " / " + imagenes.length;
                }
            }
        }

        function mostrarSiguiente() {
            indiceActual = (indiceActual + 1) % imagenes.length;
            actualizarImagen();
        }

        function mostrarAnterior() {
            indiceActual = (indiceActual - 1 + imagenes.length) % imagenes.length;
            actualizarImagen();
        }

        function cerrarModal() {
            lightbox.style.display = "none";
            document.body.style.overflow = "auto";
        }

        if (closeBtn) closeBtn.addEventListener("click", cerrarModal);
        if (nextBtn) nextBtn.addEventListener("click", mostrarSiguiente);
        if (prevBtn) prevBtn.addEventListener("click", mostrarAnterior);
        
        lightbox.addEventListener("click", function (evento) {
            if (evento.target === lightbox) cerrarModal();
        });

        document.addEventListener("keydown", function (evento) {
            if (lightbox.style.display === "block") {
                if (evento.key === "ArrowRight") mostrarSiguiente();
                if (evento.key === "ArrowLeft") mostrarAnterior();
                if (evento.key === "Escape") cerrarModal();
            }
        });
    }

    if (document.readyState === "loading") {
        document.addEventListener("DOMContentLoaded", inicializarLightbox);
    } else {
        inicializarLightbox();
    }
})();
</script>

async function buscarCP(cp) {
  if (cp.length !== 5) {
    document.getElementById('cp-fields').style.display = 'none';
    document.getElementById('cp-error').style.display = 'none';
    return;
  }
  document.getElementById('cp-loading').style.display = 'block';
  document.getElementById('cp-fields').style.display = 'none';
  document.getElementById('cp-error').style.display = 'none';

  try {
    const res = await fetch(`https://postali.app/api/v1/cp/${cp}`);
    const data = await res.json();
    document.getElementById('cp-loading').style.display = 'none';

    if (!data.asentamientos || data.asentamientos.length === 0) {
      document.getElementById('cp-error').style.display = 'block';
      return;
    }

    document.getElementById('f-ciudad').value = data.municipio || '';

    const select = document.getElementById('f-colonia');
    select.innerHTML = '';
    data.asentamientos.forEach(z => {
      const opt = document.createElement('option');
      opt.value = z.nombre;
      opt.textContent = z.nombre;
      select.appendChild(opt);
    });

    document.getElementById('cp-fields').style.display = 'grid';

  } catch(e) {
    document.getElementById('cp-loading').style.display = 'none';
    document.getElementById('cp-error').style.display = 'block';
  }
}
